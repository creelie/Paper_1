  vertex/.style={circle,draw=black,fill=white,inner sep=0pt,minimum size=5pt}]

  \coordinate (Y\i) at (\i,0);}

  \node[vertex] at (X\i) {};
  \node[vertex] at (Y\i) {};
  \node[above=5pt] at (X\i) {$X_{\i}$};
  \node[below=5pt] at (Y\i) {$Y_{\i}$};}
